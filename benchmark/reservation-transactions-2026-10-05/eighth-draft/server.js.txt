#!/usr/bin/env node
'use strict';
const http=require('node:http'),fs=require('node:fs'),path=require('node:path');const {DatabaseSync}=require('node:sqlite');
function arg(n){let i=process.argv.indexOf(n);return i<0?null:process.argv[i+1]}const dbFile=arg('--db')||process.env.DB||'.atoma-scratch/live.sqlite',seedFile=arg('--seed')||process.env.SEED||'.atoma-scratch/live-seed.json',port=Number(arg('--port')??process.env.PORT);
if(!dbFile||!seedFile||!Number.isInteger(port)||port<0||port>65535){console.error('usage: node server.js --db DB --seed SEED.json --port PORT');process.exit(2)}
const obj=x=>x!==null&&typeof x==='object'&&!Array.isArray(x), nb=x=>typeof x==='string'&&x.length>0, int=(x,a,b)=>Number.isInteger(x)&&x>=a&&x<=b;
const exact=(o,k)=>obj(o)&&Object.keys(o).length===k.length&&k.every(x=>Object.prototype.hasOwnProperty.call(o,x)), cmp=(a,b)=>a<b?-1:a>b?1:0;
function canon(v){if(Array.isArray(v))return '['+v.map(canon).join(',')+']';if(obj(v))return '{'+Object.keys(v).sort(cmp).map(k=>JSON.stringify(k)+':'+canon(v[k])).join(',')+'}';return JSON.stringify(v)}
let seed;try{seed=JSON.parse(fs.readFileSync(seedFile,'utf8'))}catch{console.error('invalid seed');process.exit(2)}
if(!exact(seed,['items'])||!Array.isArray(seed.items)){console.error('invalid seed');process.exit(2)}
const sk=new Set;for(const x of seed.items)if(!exact(x,['sku','capacity'])||!nb(x.sku)||sk.has(x.sku)||!int(x.capacity,0,1000000)){console.error('invalid seed');process.exit(2)}else sk.add(x.sku);
fs.mkdirSync(path.dirname(path.resolve(dbFile)),{recursive:true});const db=new DatabaseSync(dbFile);db.exec('PRAGMA journal_mode=WAL; PRAGMA synchronous=FULL;');
db.exec('CREATE TABLE IF NOT EXISTS meta (id INTEGER PRIMARY KEY CHECK(id=1),clock INTEGER NOT NULL,revision INTEGER NOT NULL);CREATE TABLE IF NOT EXISTS items(sku TEXT PRIMARY KEY,capacity INTEGER NOT NULL,available INTEGER NOT NULL);CREATE TABLE IF NOT EXISTS holds(id TEXT PRIMARY KEY,expiresAt INTEGER NOT NULL,status TEXT NOT NULL);CREATE TABLE IF NOT EXISTS lines(holdId TEXT NOT NULL,sku TEXT NOT NULL,qty INTEGER NOT NULL,PRIMARY KEY(holdId,sku));CREATE TABLE IF NOT EXISTS receipts(key TEXT PRIMARY KEY,command TEXT NOT NULL,status INTEGER NOT NULL,body TEXT NOT NULL)');
if(!db.prepare('SELECT id FROM meta WHERE id=1').get()){db.exec('BEGIN IMMEDIATE');try{db.prepare('INSERT INTO meta VALUES(1,0,0)').run();let q=db.prepare('INSERT INTO items VALUES(?,?,?)');for(const x of seed.items)q.run(x.sku,x.capacity,x.capacity);db.exec('COMMIT')}catch(e){db.exec('ROLLBACK');throw e}}
function state(){let m=db.prepare('SELECT clock,revision FROM meta').get(),items=db.prepare('SELECT sku,capacity,available FROM items').all().sort((a,b)=>cmp(a.sku,b.sku)),hs=db.prepare('SELECT id,expiresAt,status FROM holds').all().sort((a,b)=>cmp(a.id,b.id)),ls=db.prepare('SELECT holdId,sku,qty FROM lines').all();return{clock:m.clock,revision:m.revision,items,holds:hs.map(h=>({id:h.id,expiresAt:h.expiresAt,status:h.status,lines:ls.filter(x=>x.holdId===h.id).sort((a,b)=>cmp(a.sku,b.sku)).map(x=>({sku:x.sku,qty:x.qty}))}))}}
function valid(c){if(!obj(c)||!nb(c.key)||!nb(c.type))return false;if(c.type==='reserve')return exact(c,['key','type','holdId','ttl','lines'])&&nb(c.holdId)&&int(c.ttl,1,1e6)&&Array.isArray(c.lines)&&c.lines.length>0&&c.lines.every(x=>exact(x,['sku','qty'])&&nb(x.sku)&&int(x.qty,1,1e6));if(c.type==='confirm'||c.type==='cancel')return exact(c,['key','type','holdId'])&&nb(c.holdId);return c.type==='advance'&&exact(c,['key','type','to'])&&int(c.to,0,1e9)}
function biz(c){let m=db.prepare('SELECT clock FROM meta').get();if(c.type==='advance')return c.to<m.clock?'clock_backwards':null;if(c.type==='reserve'){if(db.prepare('SELECT id FROM holds WHERE id=?').get(c.holdId))return'hold_exists';let sums=Object.create(null);for(let x of c.lines)sums[x.sku]=Math.min(1000001,(sums[x.sku]||0)+x.qty);for(let sku of Object.keys(sums)){let i=db.prepare('SELECT available FROM items WHERE sku=?').get(sku);if(!i)return'missing_item';if(i.available<sums[sku])return'insufficient_stock'}return null}let h=db.prepare('SELECT status FROM holds WHERE id=?').get(c.holdId);if(!h)return'missing_hold';if(h.status!=='held')return'inactive_hold';if(c.type==='advance'&&c.to<m.clock)return'clock_backwards';return null}
function execute(c){let cmd=JSON.parse(JSON.stringify(c));delete cmd.key;let old=db.prepare('SELECT command,status,body FROM receipts WHERE key=?').get(c.key);if(old)return old.command===canon(cmd)?{status:old.status,body:old.body}:{status:409,body:'{"error":"idempotency_conflict"}'};db.exec('BEGIN IMMEDIATE');try{let e=biz(c),status=200,body;if(e){status=409;body=JSON.stringify({error:e})}else{if(c.type==='reserve'){let s=Object.create(null);for(let x of c.lines)s[x.sku]=(s[x.sku]||0)+x.qty;for(let [k,v]of Object.entries(s))db.prepare('UPDATE items SET available=available-? WHERE sku=?').run(v,k);let clock=db.prepare('SELECT clock FROM meta').get().clock;db.prepare('INSERT INTO holds VALUES(?,?,?)').run(c.holdId,clock+c.ttl,'held');for(let[k,v]of Object.entries(s))db.prepare('INSERT INTO lines VALUES(?,?,?)').run(c.holdId,k,v)}else if(c.type==='confirm')db.prepare("UPDATE holds SET status='confirmed' WHERE id=?").run(c.holdId);else if(c.type==='cancel'){for(let x of db.prepare('SELECT sku,qty FROM lines WHERE holdId=?').all(c.holdId))db.prepare('UPDATE items SET available=available+? WHERE sku=?').run(x.qty,x.sku);db.prepare("UPDATE holds SET status='cancelled' WHERE id=?").run(c.holdId)}else{db.prepare('UPDATE meta SET clock=?').run(c.to);for(let h of db.prepare("SELECT id FROM holds WHERE status='held' AND expiresAt<=?").all(c.to)){for(let x of db.prepare('SELECT sku,qty FROM lines WHERE holdId=?').all(h.id))db.prepare('UPDATE items SET available=available+? WHERE sku=?').run(x.qty,x.sku);db.prepare("UPDATE holds SET status='expired' WHERE id=?").run(h.id)}}let r=db.prepare('UPDATE meta SET revision=revision+1 RETURNING revision').get().revision;body=JSON.stringify({ok:true,revision:r})}db.prepare('INSERT INTO receipts VALUES(?,?,?,?)').run(c.key,canon(cmd),status,body);db.exec('COMMIT');return{status,body}}catch(e){try{db.exec('ROLLBACK')}catch{}throw e}}
let queue=Promise.resolve();function serial(f){let p=queue.then(f);queue=p.catch(()=>{});return p}
const srv=http.createServer((req,res)=>{if(req.method==='GET'&&req.url==='/state'){let b=JSON.stringify(state());res.writeHead(200,{'content-type':'application/json'});return res.end(b)}if(req.method!=='POST'||req.url!=='/commands'){res.writeHead(404);return res.end()}let raw='';req.on('data',d=>raw+=d);req.on('end',()=>{let c;try{c=JSON.parse(raw)}catch{return res.writeHead(400,{'content-type':'application/json'}),res.end('{"error":"invalid"}')}if(!valid(c))return res.writeHead(400,{'content-type':'application/json'}),res.end('{"error":"invalid"}');serial(()=>execute(c)).then(x=>{res.writeHead(x.status,{'content-type':'application/json'});res.end(x.body)}).catch(()=>{res.writeHead(500);res.end()})})});srv.listen(port,'127.0.0.1',()=>console.log(JSON.stringify({port:srv.address().port})));
