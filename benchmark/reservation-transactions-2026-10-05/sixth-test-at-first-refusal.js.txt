const assert=require('node:assert/strict'),{spawn}=require('node:child_process'),fs=require('node:fs'),os=require('node:os'),path=require('node:path'),http=require('node:http');
const root=__dirname;
function make(items){const d=fs.mkdtempSync(path.join(os.tmpdir(),'stock-'));fs.writeFileSync(path.join(d,'seed.json'),JSON.stringify({items}));return {db:path.join(d,'x.db'),seed:path.join(d,'seed.json')}}
function start(x){return new Promise((resolve,reject)=>{const p=spawn(process.execPath,[path.join(root,'server.js'),'--db',x.db,'--seed',x.seed,'--port','0']);let b='';p.stdout.on('data',d=>{b+=d;const m=b.match(/\{"port":(\d+)\}/);if(m)resolve({p,port:+m[1]})});p.on('error',reject)})}
function stop(s){return new Promise(r=>{s.p.once('exit',r);s.p.kill('SIGTERM')})}
function call(port,body,raw=false){return new Promise((resolve,reject)=>{const q=http.request({host:'127.0.0.1',port,method:'POST',path:'/commands',headers:{'content-type':'application/json'}},r=>{let b='';r.on('data',d=>b+=d);r.on('end',()=>resolve({status:r.statusCode,body:b,json:()=>JSON.parse(b)}))});q.on('error',reject);q.end(raw?body:JSON.stringify(body))})}
function state(port){return new Promise((resolve,reject)=>{http.get({host:'127.0.0.1',port,path:'/state'},r=>{let b='';r.on('data',d=>b+=d);r.on('end',()=>resolve({body:b,json:()=>JSON.parse(b)}))}).on('error',reject)})}
