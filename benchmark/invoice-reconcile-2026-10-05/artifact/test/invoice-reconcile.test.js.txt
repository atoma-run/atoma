'use strict';
const assert=require('assert'),fs=require('fs'),os=require('os'),path=require('path'),cp=require('child_process');
const root=path.resolve(__dirname,'..'), cli=path.join(root,'invoice-reconcile.js'), dir=fs.mkdtempSync(path.join(os.tmpdir(),'reconcile-'));
function run(args){return cp.spawnSync(process.execPath,[cli,...args],{encoding:'utf8'})}
function write(name,x){let p=path.join(dir,name);fs.writeFileSync(p,JSON.stringify(x));return p}
function go(name,x,out='out.json'){let i=write(name,x),o=path.join(dir,out),r=run(['--input',i,'--out',o]);assert.equal(r.status,0,r.stderr);return {v:JSON.parse(fs.readFileSync(o)),b:fs.readFileSync(o)}}
let x={invoices:[{id:'E',currency:'EUR',lines:[{quantity:3,unitPrice:'1.005'}],taxBps:2000},{id:'j',currency:'JPY',lines:[{quantity:3,unitPrice:'10.5'}],taxBps:1000},{id:'k',currency:'KWD',lines:[{quantity:3,unitPrice:'1.2345'}],taxBps:500},{id:'u',currency:'USD',lines:[{quantity:2,unitPrice:'9007199254740991.99'}],taxBps:0}],credits:[{id:'c',invoiceId:'E',currency:'EUR',amount:'0.62'}],payments:[{id:'p1',currency:'EUR',amount:'2.00',allocations:[{invoiceId:'E',amount:'1.10'}]},{id:'p2',currency:'EUR',amount:'1.90',allocations:[{invoiceId:'E',amount:'1.90'}]},{id:'z',currency:'EUR',amount:'0.00',allocations:[]}]};
let r=go('ledger.json',x);assert.deepEqual(r.v.invoices[0],{id:'E',currency:'EUR',subtotal:'3.02',tax:'0.60',total:'3.62',credited:'0.62',paid:'3.00',balance:'0.00',status:'settled'});assert.equal(r.v.invoices.find(v=>v.id==='j').total,'35');assert.equal(r.v.invoices.find(v=>v.id==='k').total,'3.889');assert.equal(r.v.invoices.find(v=>v.id==='u').subtotal,'18014398509481983.98');assert.equal(r.v.payments.find(v=>v.id==='p1').unapplied,'0.90');
let dup={invoices:[{id:'__proto__',currency:'EUR',lines:[{quantity:1,unitPrice:'0.00'}],taxBps:0},{currency:'EUR',taxBps:0,lines:[{unitPrice:'0.00',quantity:1}],id:'__proto__'}],credits:[{id:'c',invoiceId:'__proto__',currency:'EUR',amount:'0.00'}],payments:[{id:'c',currency:'EUR',amount:'0.00',allocations:[]},{currency:'EUR',allocations:[],amount:'0.00',id:'c'}]};assert.equal(go('dup.json',dup).v.duplicates.invoices,1);assert.equal(go('dup2.json',{...dup,invoices:[...dup.invoices,{...dup.invoices[0],id:'a'}]}).v.invoices.length,2);
let conflict={invoices:[{id:'x',currency:'EUR',lines:[{quantity:1,unitPrice:'1.00'}],taxBps:0}],credits:[{id:'c',invoiceId:'x',currency:'EUR',amount:'0.00'},{invoiceId:'x',id:'c',currency:'EUR',amount:'1.00'}],payments:[]};assert.notEqual(run(['--input',write('conflict.json',conflict),'--out',path.join(dir,'conflict-out')]).status,0);
let base=path.join(dir,'prior');fs.writeFileSync(base,'KEEP');let bad=write('bad.json',{invoices:[],credits:[{id:'x',invoiceId:'missing',currency:'EUR',amount:'1.00'}],payments:[]});let q=run(['--input',bad,'--out',base]);assert.notEqual(q.status,0);assert.equal(fs.readFileSync(base,'utf8'),'KEEP');
for(const badx of [{invoices:[],credits:[],payments:[],extra:1},{invoices:[{id:'x',currency:'EUR',lines:[],taxBps:0}],credits:[],payments:[]},{invoices:[],credits:[],payments:[{id:'p',currency:'EUR',amount:'1.00',allocations:[{invoiceId:'x',amount:'1.00'}]}]}]){let z=write('badx.json',badx);assert.notEqual(run(['--input',z,'--out',path.join(dir,'q')]).status,0)}
function refusal(name,data){let i=write(name+'.json',data),o=path.join(dir,name+'.out');fs.writeFileSync(o,'KEEP');let z=run(['--input',i,'--out',o]);assert.notEqual(z.status,0,name+' should refuse');assert.ok(z.stderr.trim().length>0,name+' should diagnose on stderr');assert.equal(fs.readFileSync(o,'utf8'),'KEEP',name+' must preserve output bytes')}
const oneInvoice={id:'i',currency:'EUR',lines:[{quantity:1,unitPrice:'10.00'}],taxBps:0};
refusal('missing-reference',{invoices:[oneInvoice],credits:[{id:'c',invoiceId:'no-such-invoice',currency:'EUR',amount:'1.00'}],payments:[]});
refusal('credit-currency-mismatch',{invoices:[oneInvoice],credits:[{id:'c',invoiceId:'i',currency:'USD',amount:'1.00'}],payments:[]});
refusal('payment-currency-mismatch',{invoices:[oneInvoice],credits:[],payments:[{id:'p',currency:'USD',amount:'1.00',allocations:[{invoiceId:'i',amount:'1.00'}]}]});
refusal('credit-exceeds-balance',{invoices:[oneInvoice],credits:[{id:'c',invoiceId:'i',currency:'EUR',amount:'10.01'}],payments:[]});
refusal('allocations-exceed-payment',{invoices:[oneInvoice],credits:[],payments:[{id:'p',currency:'EUR',amount:'1.00',allocations:[{invoiceId:'i',amount:'0.60'},{invoiceId:'i',amount:'0.50'}]}]});
refusal('allocation-exceeds-invoice',{invoices:[oneInvoice],credits:[],payments:[{id:'p',currency:'EUR',amount:'20.00',allocations:[{invoiceId:'i',amount:'10.01'}]}]});
let same=write('same.json',{invoices:[],credits:[],payments:[]});assert.notEqual(run(['--input',same,'--out',same]).status,0);assert.equal(run(['--help']).status,0);
let hard=path.join(dir,'hard');fs.linkSync(same,hard);assert.notEqual(run(['--input',same,'--out',hard]).status,0);let sym=path.join(dir,'sym');fs.symlinkSync(same,sym);assert.notEqual(run(['--input',same,'--out',sym]).status,0);assert.equal(go('empty.json',{invoices:[],credits:[],payments:[]},'distinct.json').v.duplicates.invoices,0);
console.log('invoice-reconcile tests passed');
