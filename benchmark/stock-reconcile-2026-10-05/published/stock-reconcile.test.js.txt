const test=require('node:test');
const assert=require('node:assert/strict');
const fs=require('node:fs'), os=require('node:os'), path=require('node:path');
const {spawnSync}=require('node:child_process');
const cli=path.resolve(__dirname,'..','stock-reconcile.js');
function run(dir,args){return spawnSync(process.execPath,[cli,...args],{cwd:dir,encoding:'utf8'});}
function setup(open='sku,onHand\nA,10\nB,0\n', events='id,sku,type,quantity,ref\n', files={}) {
 const dir=fs.mkdtempSync(path.join(os.tmpdir(),'stock-')); fs.writeFileSync(path.join(dir,'opening.csv'),open); fs.writeFileSync(path.join(dir,'events.csv'),events);
 for(const [n,v] of Object.entries(files))fs.writeFileSync(path.join(dir,n),v); return dir;
}
test('--help',()=>{const r=run(process.cwd(),['--help']);assert.equal(r.status,0);assert.match(r.stdout,/Usage:/);});
test('fixed example and deterministic repeat',()=>{
 const d=setup(); fs.copyFileSync(path.join(__dirname,'..','examples/opening.csv'),path.join(d,'fixed-opening.csv')); fs.copyFileSync(path.join(__dirname,'..','examples/events.csv'),path.join(d,'fixed-events.csv'));
 let r=run(d,['--opening','fixed-opening.csv','--events','fixed-events.csv','--out','out']);assert.equal(r.status,0,r.stderr);
 const first=fs.readFileSync(path.join(d,'out','stock.csv'),'utf8'); const audit=JSON.parse(fs.readFileSync(path.join(d,'out','audit.json'))); assert.equal(first,'sku,onHand,reserved,available\nA,9,0,9\nB,7,2,5\n');assert.equal(audit.applied,6);assert.equal(audit.duplicates,1);
 r=run(d,['--opening','fixed-opening.csv','--events','fixed-events.csv','--out','out']);assert.equal(r.status,0);assert.equal(fs.readFileSync(path.join(d,'out','stock.csv'),'utf8'),first);
});
test('partial transitions and repeated shipment',()=>{
 const d=setup('sku,onHand\nA,10\n','id,sku,type,quantity,ref\nr,A,RESERVE,6,R\ns,A,SHIP,4,R\ns,A,SHIP,4,R\nl,A,RELEASE,2,R\nt,A,RETURN,3,s\n');
 assert.equal(run(d,['--opening','opening.csv','--events','events.csv','--out','o']).status,0);
 const a=JSON.parse(fs.readFileSync(path.join(d,'o','audit.json')));assert.equal(a.duplicates,1);assert.equal(a.events.at(-1).after.onHand,9);
});
test('csv hostile and CRLF BOM',()=>{
 const d=setup('\ufeffsku,onHand\r\nA,2\r\n','id,sku,type,quantity,ref\r\n"e,1","A",RECEIVE,1,""\r\n');
 const r=run(d,['--opening','opening.csv','--events','events.csv','--out','o']);assert.equal(r.status,0,r.stderr);assert.match(fs.readFileSync(path.join(d,'o','stock.csv'),'utf8'),/sku,onHand.*\nA,3/);
});
test('multiline and doubled quote identifiers',()=>{
 const d=setup('sku,onHand\n"weird\nsku",2\n','id,sku,type,quantity,ref\n"x""1","weird\nsku",RECEIVE,1,\n');
 const r=run(d,['--opening','opening.csv','--events','events.csv','--out','o']);assert.equal(r.status,0,r.stderr);assert.match(fs.readFileSync(path.join(d,'o','stock.csv'),'utf8'),/"weird\nsku",3,0,3/);
});
test('conflicts, prerequisites, malformed and quantities fail; outputs preserved',()=>{
 const cases=[
  'id,sku,type,quantity,ref\nx,A,RECEIVE,1,bad\n',
  'id,sku,type,quantity,ref\nx,A,RESERVE,11,r\n',
  'id,sku,type,quantity,ref\nx,A,RELEASE,1,r\n',
  'id,sku,type,quantity,ref\nx,A,RETURN,1,s\n',
  'id,sku,type,quantity,ref\nx,A,RECEIVE,nope,\n',
  'id,sku,type,quantity,ref\nx,A,RECEIVE,1,"bad\n',
  'id,sku,type,quantity,ref\nx,A,RECEIVE,1,\nextra\n',
  'id,sku,type,quantity,ref\nx,A,RECEIVE,1,\nlate,A,RECEIVE,9007199254740992,\n'
 ];
 for(const ev of cases){const d=setup('sku,onHand\nA,1\n',ev);fs.mkdirSync(path.join(d,'o'));fs.writeFileSync(path.join(d,'o','stock.csv'),'keep');fs.writeFileSync(path.join(d,'o','audit.json'),'keep2');const r=run(d,['--opening','opening.csv','--events','events.csv','--out','o']);assert.notEqual(r.status,0);assert.match(r.stderr,/row|input|quantity/);assert.equal(fs.readFileSync(path.join(d,'o','stock.csv'),'utf8'),'keep');assert.equal(fs.readFileSync(path.join(d,'o','audit.json'),'utf8'),'keep2');}
});
test('opening-only, overwrite refusal, excess return',()=>{
 const d=setup('sku,onHand\nZ,0\n','id,sku,type,quantity,ref\n');let r=run(d,['--opening','opening.csv','--events','events.csv','--out','o']);assert.equal(r.status,0);
 r=run(d,['--opening','opening.csv','--events','events.csv','--out','opening.csv']);assert.notEqual(r.status,0);
 const e=path.join(d,'bad.csv');fs.copyFileSync(path.join(__dirname,'..','examples/excess-return-events.csv'),e);fs.copyFileSync(path.join(__dirname,'..','examples/opening.csv'),path.join(d,'fo.csv'));fs.mkdirSync(path.join(d,'ex'));r=run(d,['--opening','fo.csv','--events',e,'--out','ex']);assert.notEqual(r.status,0);assert.equal(fs.readdirSync(path.join(d,'ex')).length,0);
});
