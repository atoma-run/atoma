function refusalPreserves(name,data){
  const i=write(name+'.json',data), before=fs.readFileSync(i);
  const o=path.join(dir,name+'.out');fs.writeFileSync(o,'PREEXISTING\\n');const outBefore=fs.readFileSync(o);
  const z=run(['--input',i,'--out',o]);
  assert.notEqual(z.status,0,name+' should fail');
  assert.ok(z.stderr.trim().length>0,name+' should diagnose');
  assert.deepEqual(fs.readFileSync(i),before,name+' must preserve input bytes');
  assert.deepEqual(fs.readFileSync(o),outBefore,name+' must preserve output bytes');
}
